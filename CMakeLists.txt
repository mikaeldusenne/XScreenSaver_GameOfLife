cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" LIFE_VERSION LIMIT_COUNT 1)
project(GameOfLife VERSION ${LIFE_VERSION} LANGUAGES C)

if(NOT UNIX)
  message(FATAL_ERROR "Native executables need a POSIX system. On Windows, use WSL; the duel client also runs in a browser. See docs/INSTALL.md.")
endif()
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(LIFE_WITH_XSS "Build the optional X11 idle screensaver" ON)
option(LIFE_WITH_GPU "Build the optional OpenGL 4.3 engine when libepoxy is available" ON)
option(LIFE_WITH_X11 "Build the X11 screensaver and wallpaper" ON)
option(LIFE_WITH_DUEL "Build the optional browser duel server when libwebsockets is available" ON)
option(BUILD_TESTING "Build the existing simulation checks" OFF)
include(GNUInstallDirs)
find_package(PkgConfig QUIET)
add_library(life_core STATIC life.c figures.c scene.c journal.c)
target_include_directories(life_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(LIFE_WITH_X11)
  find_package(X11 REQUIRED)
  if(LIFE_WITH_XSS AND NOT X11_Xss_FOUND)
    message(STATUS "libXss not found: preview, wallpaper and XScreenSaver integration remain available; --idle needs libXss")
  endif()
  add_executable(game_of_life game_of_life.c gpu.c)
  target_link_libraries(game_of_life PRIVATE life_core X11::X11)
  if(LIFE_WITH_GPU AND NOT APPLE AND PkgConfig_FOUND)
    pkg_check_modules(EPOXY QUIET IMPORTED_TARGET epoxy)
  endif()
  if(LIFE_WITH_GPU AND NOT APPLE AND EPOXY_FOUND)
    target_link_libraries(game_of_life PRIVATE PkgConfig::EPOXY)
    target_compile_definitions(game_of_life PRIVATE LIFE_HAVE_GPU=1)
  else()
    target_compile_definitions(game_of_life PRIVATE LIFE_HAVE_GPU=0)
    message(STATUS "Building CPU renderer; optional --gpu needs libepoxy and OpenGL 4.3 at runtime")
  endif()
  target_compile_definitions(game_of_life PRIVATE LIFE_VERSION="${PROJECT_VERSION}")
  if(LIFE_WITH_XSS AND X11_Xss_FOUND)
    target_link_libraries(game_of_life PRIVATE X11::Xss)
    target_compile_definitions(game_of_life PRIVATE LIFE_HAVE_XSS=1)
  else()
    target_compile_definitions(game_of_life PRIVATE LIFE_HAVE_XSS=0)
  endif()
endif()

if(LIFE_WITH_DUEL AND PkgConfig_FOUND)
  pkg_check_modules(LWS QUIET IMPORTED_TARGET libwebsockets)
endif()
if(LIFE_WITH_DUEL AND LWS_FOUND)
  add_executable(life-duel-server duel.c duel_server.c)
  target_link_libraries(life-duel-server PRIVATE life_core PkgConfig::LWS)
  target_compile_definitions(life-duel-server PRIVATE
    LIFE_DUEL_WEB_DIR="${CMAKE_INSTALL_FULL_DATADIR}/game-of-life/web"
    LIFE_DUEL_WEB_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/web")
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(life-duel-server PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  install(TARGETS life-duel-server RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(DIRECTORY web/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/game-of-life/web)
elseif(LIFE_WITH_DUEL)
  message(STATUS "Optional browser duel server needs libwebsockets development files")
  if(NOT LIFE_WITH_X11)
    message(FATAL_ERROR "Server-only build requires libwebsockets")
  endif()
endif()

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(life_core PRIVATE -Wall -Wextra -Wpedantic)
  if(LIFE_WITH_X11)
    target_compile_options(game_of_life PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

if(BUILD_TESTING)
  enable_testing()
  foreach(check life figures scene)
    add_executable(${check}_test tests/${check}_test.c)
    target_link_libraries(${check}_test PRIVATE life_core)
    # These existing checks use assert(), including setup with side effects.
    target_compile_options(${check}_test PRIVATE -UNDEBUG)
    add_test(NAME ${check} COMMAND ${check}_test)
  endforeach()
endif()

# Desktop entries have two escaping layers. Quote paths, including spaces.
if(LIFE_WITH_X11)
  set(LIFE_DESKTOP_EXEC "${CMAKE_INSTALL_FULL_BINDIR}/game_of_life")
  string(REPLACE "\\" "\\\\\\\\" LIFE_DESKTOP_EXEC "${LIFE_DESKTOP_EXEC}")
  string(REPLACE "\"" "\\\\\"" LIFE_DESKTOP_EXEC "${LIFE_DESKTOP_EXEC}")
  string(REPLACE "`" "\\\\`" LIFE_DESKTOP_EXEC "${LIFE_DESKTOP_EXEC}")
  string(REPLACE "$" "\\\\$" LIFE_DESKTOP_EXEC "${LIFE_DESKTOP_EXEC}")
  string(REPLACE "%" "%%" LIFE_DESKTOP_EXEC "${LIFE_DESKTOP_EXEC}")
  configure_file(contrib/game-of-life.desktop.in game-of-life.desktop @ONLY)
  install(TARGETS game_of_life RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/game-of-life.desktop"
          DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/applications)
endif()
install(PROGRAMS scripts/game-of-life DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES contrib/game-of-life.svg
        DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/icons/hicolor/scalable/apps)
install(FILES contrib/game-of-life.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES README.md LICENSE FIGURES.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
install(DIRECTORY ci/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/ci)
install(FILES bench/README.md bench/results.png bench/PERFORMANCE.md
        bench/compute-2026-10-27.json bench/render-2026-10-27.json
        bench/DENSE.md bench/dense-compute-2026-10-27.json
        bench/dense-metadata-2026-10-27.json bench/dense-baseline-2026-10-27.json
        bench/dense-render-2026-10-27.json bench/dense-mix-2026-10-27.json
        bench/GPU.md bench/gpu-render-2026-10-27.json
        bench/rounded-2026-10-27.json
        bench/fancy-draw-2026-10-28.json bench/fancy-render-2026-10-28.json
        bench/duel-2026-10-28.json
        DESTINATION ${CMAKE_INSTALL_DOCDIR}/bench)
install(FILES img/preview.png img/figure-pulsar.png img/figure-koks-galaxy.png
        img/figure-bunnies.png img/fancy-pulsar.png DESTINATION ${CMAKE_INSTALL_DOCDIR}/img)
install(FILES contrib/game-of-life.svg DESTINATION ${CMAKE_INSTALL_DOCDIR}/contrib)
install(FILES contrib/game-of-life-wallpaper.service DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/game-of-life)
install(FILES contrib/completions/game-of-life.bash
        DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/bash-completion/completions RENAME game-of-life)
configure_file(cmake/uninstall.cmake.in uninstall.cmake @ONLY)
add_custom_target(uninstall COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")

# Native archives/packages can be built with cpack; no privileged installation.
set(CPACK_PACKAGE_NAME game-of-life)
set(CPACK_PACKAGE_VENDOR "Game of Life contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "A small, colorful X11 Life screensaver with 810 classic patterns")
set(CPACK_PACKAGE_CONTACT "https://github.com/mikaeldusenne/XScreenSaver_GameOfLife/issues")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/mikaeldusenne/XScreenSaver_GameOfLife")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGING_INSTALL_PREFIX "${CMAKE_INSTALL_PREFIX}")
set(CPACK_PACKAGE_FILE_NAME "game-of-life-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_RPM_PACKAGE_LICENSE "LicenseRef-GameOfLife")
include(CPack)
